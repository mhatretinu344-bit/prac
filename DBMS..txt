/

Package Created.
-- 4. Trigger 1: Student -> Student_Replica
SQL> CREATE TRIGGER Std_to_StdRep
AFTER INSERT OR UPDATE OR DELETE ON Student
FOR EACH ROW
BEGIN
 IF NOT Replication_Control.is_replicating THEN
  Replication_Control.is_replicating := TRUE;

  IF INSERTING THEN
   INSERT INTO Student_Replica (Id, Name, City)
   VALUES (:NEW.Id, :NEW.Name, :NEW.City);
  ELSIF UPDATING THEN
   UPDATE Student_Replica
   SET Name = :NEW.Name, City = :NEW.City
   WHERE Id = :OLD.Id;
  ELSE
   DELETE FROM Student_Replica
   WHERE Id = :OLD.Id;
  END IF;

  Replication_Control.is_replicating := FALSE;
 END IF;
END;
/

Trigger Created.
-- 5. Trigger 2: Student_Replica -> Student
SQL> CREATE TRIGGER StdRep_to_Std
AFTER INSERT OR UPDATE OR DELETE ON Student_Replica
FOR EACH ROW
BEGIN
 IF NOT Replication_Control.is_replicating THEN
  Replication_Control.is_replicating := TRUE;

  IF INSERTING THEN
   INSERT INTO Student (Id, Name, City)

  ELSIF UPDATING THEN

   WHERE Id = :OLD.Id;
  ELSIF DELETING THEN
   DELETE FROM Student
   WHERE Id = :OLD.Id;

CREATE OR REPLACE TRIGGER trg_bank_temporal
AFTER INSERT OR UPDATE OR DELETE ON bank_account
FOR EACH ROW
BEGIN
 IF INSERTING THEN
  INSERT INTO bank_account_history (account_no, customer_name, balance, action_performed)
  VALUES (:NEW.account_no, :NEW.customer_name, :NEW.balance, 'INSERT');

 ELSIF UPDATING THEN
  INSERT INTO bank_account_history (account_no, customer_name, balance, action_performed)
  VALUES (:OLD.account_no, :OLD.customer_name, :OLD.balance, 'BEFORE_UPDATE');

 ELSIF DELETING THEN
  INSERT INTO bank_account_history (account_no, customer_name, balance, action_performed)
  VALUES (:OLD.account_no, :OLD.customer_name, :OLD.balance, 'DELETE');
 END IF;
END;
